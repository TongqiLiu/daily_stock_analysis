{VCP_H1_H2_BUY 日线版 主图指标 · 优化版}
{ Minervini VCP: 趋势模板 + 波动/量能收缩 + 枢轴突破 + H1/H2 }
{ 建议加载在日线图 }

E10:=EMA(CLOSE,10);
E21:=EMA(CLOSE,21);
E50:=EMA(CLOSE,50);
E200:=EMA(CLOSE,200);

EMA10:E10,COLORWHITE;
EMA21:E21,COLORYELLOW;
EMA50:E50,COLORBLUE;
EMA200:E200,COLORGRAY;

{ ---- 参数 ---- }
NEARP:=0.85;
RMAX:=0.20;
BUYGAP:=10;
USELONG:=1;

{一、趋势过滤(加 E50>E200 长周期确认, USELONG=0 可关)}
TBASE:=CLOSE>E21 AND E21>E50 AND E21>REF(E21,5) AND E50>REF(E50,5);
TREND:=TBASE AND (E50>E200 OR USELONG=0);

{二、高位过滤}
NEAR_HIGH:=CLOSE>=HHV(HIGH,60)*NEARP;

{三、波动收缩}
R20:=(HHV(HIGH,20)-LLV(LOW,20))/MA(CLOSE,20);
R10:=(HHV(HIGH,10)-LLV(LOW,10))/MA(CLOSE,10);
R5:=(HHV(HIGH,5)-LLV(LOW,5))/MA(CLOSE,5);
CONTRACT:=R10<R20*0.80 AND R5<R10*0.90 AND R10<RMAX;

{四、成交量持续收缩(去掉单日OR)}
DRYVOL:=MA(VOL,5)<MA(VOL,20)*0.92;

{五、回踩健康}
HOLD_EMA:=COUNT(CLOSE<E21,10)<=2 AND LLV(LOW,10)>E21*0.97;

{六、VCP准备区}
SETUP:=TREND AND NEAR_HIGH AND CONTRACT AND DRYVOL AND HOLD_EMA;
VALID_SETUP:=COUNT(SETUP,5)>=1;

{七、枢轴位(排除当日,无未来函数)}
PIVOT:=HHV(REF(HIGH,1),10);

{八、突破买点}
BREAK_BUY:=VALID_SETUP AND CLOSE>PIVOT AND CLOSE>OPEN AND VOL>MA(VOL,20)*1.10;

{九、H1}
UPTRIG:=HIGH>REF(HIGH,1) AND CLOSE>REF(CLOSE,1) AND CLOSE>OPEN;
H1:=VALID_SETUP AND UPTRIG AND REF(HIGH,1)<REF(HIGH,2) AND LOW>E21*0.97;

{十、H2(去冗余COUNT)}
H2:=VALID_SETUP AND UPTRIG AND BARSLAST(H1)>=2 AND BARSLAST(H1)<=6 AND LOW>E21*0.97;

{十一、最终买点 + 去重(BUYGAP 根内不重复)}
BUYRAW:=BREAK_BUY OR H1 OR H2;
BUY:=BUYRAW AND (COUNT(REF(BUYRAW,1),BUYGAP)=0);

{十二、图形标记(统一用 DRAWTEXT, 富途渲染稳定)}
VCP_START:=SETUP AND REF(SETUP,1)=0;
DRAWTEXT(VCP_START,LOW*0.96,'VCP'),COLORCYAN;
DRAWTEXT(H1,LOW*0.93,'H1'),COLORGREEN;
DRAWTEXT(H2,LOW*0.91,'H2'),COLORMAGENTA;
DRAWTEXT(BREAK_BUY,LOW*0.89,'BREAK'),COLORRED;
DRAWTEXT(BUY,LOW*0.985,'BUY'),COLORRED;
STICKLINE(BUY,LOW*0.99,LOW*0.985,2,0),COLORRED;

{十三、防守线与枢轴线}
STOP:E21*0.97,COLORRED;
PIVOTLINE:PIVOT,COLORCYAN;

{ ============================================================ }
{ 参数说明 }
{ NEARP   0.85  距60日高的比例下限(越大越靠高位), 0.80~0.90 }
{ RMAX    0.20  10日波动率绝对上限, 跨标的可调 0.15~0.30 }
{ BUYGAP  10    BUY去重窗口: 该根数内不重复触发 }
{ USELONG 1     1=要求 E50>E200 长周期同向; 0=关闭该过滤 }
{ }
{ 相对原版改动: }
{ 1) TREND 增加 E50>E200(带开关)  2) DRYVOL 去掉单日OR }
{ 3) H2 删冗余 COUNT(H1,6)        4) BUY 增加 BUYGAP 去重 }
{ 5) 阈值参数化  6) DRAWICON->DRAWTEXT('B')+STICKLINE(富途渲染稳定) }
{ }
{ 纪律执行(指标未内建): 止损=E21*0.97 或枢轴下方; }
{ 分批/止盈/持有规则按个人交易计划执行 }
